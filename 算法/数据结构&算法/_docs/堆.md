func build(a []int, size int) {
	for i := size/2 - 1; i >= 0; i-- {
		heapify(a, i, size)
	}
}

func heapify(a []int, i, size int) {
	l, r, largest := i*2+1, i*2+2, i
	if l < size && a[l] > a[largest] {
		largest = l
	}
	if r < size && a[r] > a[largest] {
		largest = r
	}
	if largest != i {
		a[i], a[largest] = a[largest], a[i]
		heapify(a, largest, size)
	}
}

// 插入元素到末尾, 然后自底向上确保堆的结构
func siftUp(a []int, i int) {
	parent := (i-1)/2
	if parent >=0 && a[parent] > a[i] {
		a[i], a[parent] = a[parent], a[i]
		siftUp(a, parent)
	}
}


type Interface interface {
	//sort.Interface 继承排序接口
    Len() int
    Less(i,j int) bool  // h[i] < h[j] 则为小根堆, 小的先出
    Swap(i,j int)
    // heap相关方法
    Push(x any)         // 添加元素到堆尾
    Pop() any           // 从堆尾移除元素
}

// 用法 heap.Func()
func Init(h Interface)  // 对heap进行初始化，生成小根堆（或大根堆）
func Push(h Interface, x any)  // 往堆里面插入内容
func Pop(h Interface) any  // 从堆顶pop出内容
func Remove(h Interface, i int) any  // 从指定位置删除数据，并返回删除的数据
func Fix(h Interface, i int)  // 从i位置数据发生改变后，对堆再平衡
