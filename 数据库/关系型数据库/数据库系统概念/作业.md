package org.example;  
  
import java.util.Scanner;  
import java.sql.*;  
  
public class Main {  
    public static void main(String[] args) {  
        // 配置Scanner以读取用户输入  
        Scanner scanner = new Scanner(System.in);  
        scanner.useDelimiter("\n");  
  
        // 获取用户输入的ID和密码  
        System.out.println("请输入登录ID:\n");  
        String loginID = scanner.nextLine();  
        System.out.println("请输入密码:\n");  
        String password = scanner.nextLine();  
  
        // 创建数据库连接与Statement  
        try (Connection conn = DriverManager.getConnection("jdbc:oracle:username@host:port:target_db", loginID, password);  
             Statement stmt = conn.createStatement()) {  
  
            // 循环查询至查找到有效信息  
            while (true) {  
                // 获取查询字符串  
                System.out.println("请输入查询字符串\n");  
                String search = scanner.nextLine();  
  
                // 执行查询  
                String querySQL = "SELECT ID, name FROM instructor WHERE name LIKE '%%%s%%'".formatted(search);  
                try (ResultSet rs = stmt.executeQuery(querySQL)) {  
                    if (!rs.isBeforeFirst()) {  
                        System.out.println("未找到匹配的教师，请重新输入。");  
                    } else {  
                        System.out.println("匹配的教师 (ID, name):");  
                        while (rs.next()) {  
                            int id = rs.getInt("ID");  
                            String name = rs.getString("name");  
                            System.out.println(id + ", " + name);  
                        }  
                        break;  
                    }  
                } catch (SQLException e) {  
                    System.out.println("查询失败，err msg: " + e.getMessage());  
                }  
            }  
  
            // 获取教师编号  
            System.out.println("请输入介于0到99999的ID编号\n");  
            int teacherId = -1;  
            try {  
                teacherId = Integer.parseInt(scanner.nextLine());  
                if (teacherId < 0 || teacherId > 99999) {  
                    System.out.println("ID 必须在 0 到 99999 之间，请重新输入。");  
                } else {  
                    // 检查教师是否存在  
                    String query = "SELECT ID FROM instructor WHERE ID = %d".formatted(teacherId);  
                    try (ResultSet rs = stmt.executeQuery(query);) {  
                        if (!rs.isBeforeFirst()) {  
                            System.out.println("未找到具有该 ID 的教师。");  
                            return; // 退出程序  
                        }  
                    }  
                }  
            } catch (NumberFormatException e) {  
                System.out.println("请输入有效的数字。");  
            } catch (SQLException e) {  
                System.out.println("查询失败，错误信息: " + e.getMessage());  
            }  
  
            // 获取教师信息  
            String query = "SELECT d.dept_name, c.course_id, c.title, s.sec_id, s.semester, s.year, COUNT(t.ID) AS total_students FROM teaches t JOIN section s ON t.course_id = s.course_id AND t.sec_id = s.sec_id AND t.semester = s.semester AND t.year = s.year JOIN course c ON t.course_id = c.course_id JOIN department d ON c.dept_name = d.dept_name WHERE t.ID = %d GROUP BY d.dept_name, c.course_id, s.sec_id, s.semester, s.year ORDER BY d.dept_name, c.course_id, s.year, s.semester".formatted(teacherId);  
            try (ResultSet rs = stmt.executeQuery(query)) {  
                if (!rs.isBeforeFirst()) {  
                    System.out.printf("ID为%d的教师没有教授任何课程", teacherId);  
                } else {  
                    System.out.println("教学记录:");  
                    System.out.println("系名称\t课程标识\t课程名称\t段号\t学期\t年份\t总注册人数");  
                    while (rs.next()) {  
                        String deptName = rs.getString("dept_name");  
                        String courseId = rs.getString("course_id");  
                        String title = rs.getString("title");  
                        int secId = rs.getInt("sec_id");  
                        String semester = rs.getString("semester");  
                        int year = rs.getInt("year");  
                        int totalStudents = rs.getInt("total_students");  
                        System.out.println(deptName + "\t" + courseId + "\t" + title + "\t" + secId + "\t" + semester + "\t" + year + "\t" + totalStudents);  
                    }  
                }  
            }  
        } catch (SQLException e) {  
            System.out.printf("打开数据库连接失败, err msg:%s\n重启程序以重试", e.getMessage());  
        }  
    }  
}

DELIMITER //

CREATE PROCEDURE PrintTeachingRecords(IN teacherId INT)
BEGIN
    DECLARE teacherExists INT;
    DECLARE hasCourses INT;

    -- 检查教师是否存在
    SELECT COUNT(*) INTO teacherExists
    FROM instructor
    WHERE ID = teacherId;

    IF teacherExists = 0 THEN
        SELECT '错误: 不存在具有该 ID 的教师。' AS Message;
    ELSE
        -- 检查教师是否讲授任何课程
        SELECT COUNT(*) INTO hasCourses
        FROM teaches
        WHERE ID = teacherId;

        IF hasCourses = 0 THEN
            SELECT '该教师没有讲授任何课程。' AS Message;
        ELSE
            -- 打印教师的教学记录
            SELECT 
                d.dept_name AS '系名称',
                c.course_id AS '课程标识',
                c.title AS '课程名称',
                s.sec_id AS '段号',
                s.semester AS '学期',
                s.year AS '年份',
                COUNT(t.ID) AS '总注册人数'
            FROM 
                teaches t
                JOIN section s ON t.course_id = s.course_id AND t.sec_id = s.sec_id AND t.semester = s.semester AND t.year = s.year
                JOIN course c ON t.course_id = c.course_id
                JOIN department d ON c.dept_name = d.dept_name
            WHERE 
                t.ID = teacherId
            GROUP BY 
                d.dept_name, c.course_id, s.sec_id, s.semester, s.year
            ORDER BY 
                d.dept_name, c.course_id, s.year, s.semester;
        END IF;
    END IF;
END //

DELIMITER ;
